cmake_minimum_required(VERSION 3.14)
project(http_parser LANGUAGES CXX)

include(FetchContent)

# --- Dependency: enum-reflection ---
# http_parser_enums.h does #include "enum.h" and defines its status/method/errno/
# state enums with the ENUM() macro. That header lives in the sibling
# enum-reflection library, pulled in here so the include resolves. Compiled as
# C++ (see below), enum.h's ENUM() expands to a real enum plus the reflective
# constexpr accessors, so the dependency is load-bearing, not just a path fixup.
# Pinned to a commit. FetchContent keeps the first declaration of a name, so this pin also
# decides constexpr-phf, hashes, static-string and char-classify for a consumer that
# declares none of them first.
FetchContent_Declare(
  enum_reflection
  GIT_REPOSITORY https://github.com/Kronuz/enum-reflection.git
  GIT_TAG        a7202ada8d27e7eb2d4713fbfd3b43eaa347fc91
)
FetchContent_MakeAvailable(enum_reflection)

# --- Library target ---
# http-parser is not header-only: http_parser.cc carries the ~2.4k-line state
# machine, so it must be compiled and linked. http_parser.h alone only declares
# the API. It is compiled as C++ so it pulls in enum.h's C++ machinery and
# matches the C++20 family; the file is C++-clean as written (the original
# Joyent parser uses enum casts that are already C++-valid). The implementation
# keeps its upstream content but takes a .cc extension so the C++ compile is
# first-class rather than relying on the deprecated "compile .c as C++" path.
add_library(http_parser http_parser.cc)
target_include_directories(http_parser PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(http_parser PUBLIC cxx_std_20)

# Link enum-reflection so "enum.h" is on the include path. PUBLIC because the
# public header http_parser.h -> http_parser_enums.h -> "enum.h", so consumers
# need it on their include path too.
target_link_libraries(http_parser PUBLIC enum_reflection::enum_reflection)

# Alias so consumers can link `http_parser::http_parser` whether they add this
# via add_subdirectory or FetchContent.
add_library(http_parser::http_parser ALIAS http_parser)

# Only build the test and demo when this repo is the top-level project, so
# consumers (e.g. via FetchContent) don't build them.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  enable_testing()
  add_executable(http_parser_test test/test.cc)
  target_link_libraries(http_parser_test PRIVATE http_parser)
  add_test(NAME http_parser COMMAND http_parser_test)

  # A runnable tour of the library (not a test): build & run ./build/http_parser_demo
  add_executable(http_parser_demo examples/demo.cc)
  target_link_libraries(http_parser_demo PRIVATE http_parser)

  # Differential test: the parser against test/http_parser_old.cc, a frozen copy
  # taken before the bulk-scan paths, over a corpus, mutations and random messages.
  add_executable(http_parser_diff test/diff.cc test/http_parser_old.cc)
  target_include_directories(http_parser_diff PRIVATE test)
  target_link_libraries(http_parser_diff PRIVATE http_parser)
  add_test(NAME http_parser_diff COMMAND http_parser_diff)

  # Differential fuzz target (test/fuzz.cc): the parser against the frozen copy on bytes
  # a fuzzer or a mutation loop chooses, with every split point, pause and failure.
  # http_parser_fuzz is the mutation-loop driver; a short run is a CTest test.
  add_executable(http_parser_fuzz test/fuzz.cc test/http_parser_old.cc)
  target_include_directories(http_parser_fuzz PRIVATE test)
  target_compile_definitions(http_parser_fuzz PRIVATE HTTP_PARSER_FUZZ_DRIVER)
  target_link_libraries(http_parser_fuzz PRIVATE http_parser)
  add_test(NAME http_parser_fuzz COMMAND http_parser_fuzz --iters 1500 --splits 96)

  # Conformance test: llhttp's markdown suite (test/llhttp_corpus.inc, generated by
  # tools/llhttp_corpus.mjs) against this parser, with the known divergences listed in
  # test/llhttp_divergences.txt. HTTP_PARSER_LLHTTP_SOURCE_DIR (an llhttp release tree:
  # include/llhttp.h, src/*.c) adds llhttp itself as an engine, never as a dependency of
  # the library.
  add_executable(http_parser_conformance test/conformance.cc)
  target_include_directories(http_parser_conformance PRIVATE test)
  target_link_libraries(http_parser_conformance PRIVATE http_parser)
  set(HTTP_PARSER_LLHTTP_SOURCE_DIR "" CACHE PATH "llhttp release tree for the conformance engine")
  if(HTTP_PARSER_LLHTTP_SOURCE_DIR)
    enable_language(C)
    add_library(llhttp_release STATIC
      ${HTTP_PARSER_LLHTTP_SOURCE_DIR}/src/llhttp.c
      ${HTTP_PARSER_LLHTTP_SOURCE_DIR}/src/api.c
      ${HTTP_PARSER_LLHTTP_SOURCE_DIR}/src/http.c)
    target_include_directories(llhttp_release PUBLIC ${HTTP_PARSER_LLHTTP_SOURCE_DIR}/include)
    # llhttp.h and http_parser.h define the same names, so llhttp's runner is its own file.
    target_sources(http_parser_conformance PRIVATE test/conformance_llhttp.cc)
    set_source_files_properties(test/conformance_llhttp.cc PROPERTIES INCLUDE_DIRECTORIES
      "${HTTP_PARSER_LLHTTP_SOURCE_DIR}/include;${CMAKE_CURRENT_SOURCE_DIR}/test")
    target_link_libraries(http_parser_conformance PRIVATE llhttp_release)
    target_compile_definitions(http_parser_conformance PRIVATE HAVE_LLHTTP_ENGINE)
  endif()
  add_test(NAME http_parser_conformance
    COMMAND http_parser_conformance ${CMAKE_CURRENT_SOURCE_DIR}/test/llhttp_divergences.txt)
  set_tests_properties(http_parser_conformance PROPERTIES SKIP_RETURN_CODE 77)

  # The same target as a libFuzzer binary. Configure with clang and
  # -DCMAKE_CXX_FLAGS="-fsanitize=fuzzer-no-link,address,undefined" so the parser is instrumented.
  option(HTTP_PARSER_BUILD_LIBFUZZER "Build the libFuzzer target (clang)" OFF)
  if(HTTP_PARSER_BUILD_LIBFUZZER)
    add_executable(http_parser_libfuzzer test/fuzz.cc test/http_parser_old.cc)
    target_include_directories(http_parser_libfuzzer PRIVATE test)
    target_link_libraries(http_parser_libfuzzer PRIVATE http_parser)
    target_link_options(http_parser_libfuzzer PRIVATE -fsanitize=fuzzer)
  endif()

  # A/B benchmark against the frozen copy (and llhttp when installed). Run it under
  # a quiet machine; see the header of bench/bench.cc.
  option(HTTP_PARSER_BUILD_BENCH "Build the A/B benchmark" OFF)
  if(HTTP_PARSER_BUILD_BENCH)
    add_executable(http_parser_bench bench/bench.cc test/http_parser_old.cc)
    target_include_directories(http_parser_bench PRIVATE test bench)
    target_link_libraries(http_parser_bench PRIVATE http_parser)
    find_path(LLHTTP_INCLUDE_DIR llhttp.h)
    find_library(LLHTTP_LIBRARY llhttp)
    if(LLHTTP_INCLUDE_DIR AND LLHTTP_LIBRARY)
      # llhttp.h and http_parser.h define the same names, so it gets its own source file.
      target_sources(http_parser_bench PRIVATE bench/bench_llhttp.cc)
      set_source_files_properties(bench/bench_llhttp.cc PROPERTIES INCLUDE_DIRECTORIES "${LLHTTP_INCLUDE_DIR}")
      target_compile_definitions(http_parser_bench PRIVATE HAVE_LLHTTP)
      target_link_libraries(http_parser_bench PRIVATE ${LLHTTP_LIBRARY})
    endif()

    # One workload through one parser N times: instructions per message under callgrind, or a
    # steady loop for a sampling profiler.
    add_executable(http_parser_loop bench/loop.cc test/http_parser_old.cc)
    target_include_directories(http_parser_loop PRIVATE test bench)
    target_link_libraries(http_parser_loop PRIVATE http_parser)
    if(LLHTTP_INCLUDE_DIR AND LLHTTP_LIBRARY)
      target_sources(http_parser_loop PRIVATE bench/bench_llhttp.cc)
      target_compile_definitions(http_parser_loop PRIVATE HAVE_LLHTTP)
      target_link_libraries(http_parser_loop PRIVATE ${LLHTTP_LIBRARY})
    endif()
  endif()
endif()
